# Python Program to Calculate Series
def calculate_series(n_terms):
    total = 0
    for k in range(n_terms):
        total += ((-1) ** k) / (2 * k + 1)
    return total * 4

# Calculate for the first 10,000 terms
result = calculate_series(10000)
print(result)
